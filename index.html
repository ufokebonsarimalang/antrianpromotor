function simpanAkunHP() { const namaPilih = document.getElementById('pilih-sales').value; const deviceId = getDeviceId(); if(!namaPilih) return tampilkanModal("Perhatian", "Silakan pilih nama Anda!"); tampilkanModalKonfirmasi("Kunci Perangkat", `Kunci HP ini atas nama (${namaPilih})?`, function() { const btnSimpan = document.querySelector('#panel-login button'); const textAsli = btnSimpan.innerHTML; btnSimpan.disabled = true; btnSimpan.innerHTML = ` Menyimpan...`; fetch(`${SCRIPT_URL}?aksi=kunciPerangkat&namaSales=${encodeURIComponent(namaPilih)}&deviceId=${encodeURIComponent(deviceId)}`, { method: 'GET', redirect: 'follow' }) .then(res => res.json()) .then(res => { btnSimpan.disabled = false; btnSimpan.innerHTML = textAsli; if(res.status === "sukses") { // KUNCI LANGSUNG KE MEMORI LOKAL SUPAYA SINKRON namaSalesAktifGlobal = namaPilih; localStorage.setItem('ufo_nama_sales_aktif', namaPilih); document.getElementById('panel-login').style.display = 'none'; document.getElementById('panel-menu-utama').style.display = 'block'; document.getElementById('label-user-aktif').innerText = namaPilih; tampilkanModal("Berhasil", `HP terikat untuk: ${namaPilih}`); muatDataAntrian(); } else { namaSalesAktifGlobal = ""; localStorage.removeItem('ufo_nama_sales_aktif'); document.getElementById('pilih-sales').value = ""; document.getElementById('panel-login').style.display = 'block'; document.getElementById('panel-menu-utama').style.display = 'none'; tampilkanModal("Akses Ditolak", res.pesan); } }).catch(() => { btnSimpan.disabled = false; btnSimpan.innerHTML = textAsli; tampilkanModal("Gagal", "Terjadi gangguan jaringan saat mengunci perangkat."); }); }); }
